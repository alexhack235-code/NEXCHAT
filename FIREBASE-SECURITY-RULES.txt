    // 1. Authenticated User Check
    function isAuthenticated() {
      return request.auth != null && request.auth.uid != null;
    }

    // 2. Resource Owner Check by User ID
    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    // 3. Super Admin Verification (Bootstrap email, token claim, or admins collection)
    function isSuperAdmin() {
      return isAuthenticated() && (
        (
          request.auth.token.email in ['demonalexander526@gmail.com', 'alexhack235@gmail.com'] &&
          request.auth.token.email_verified == true
        ) ||
        request.auth.token.admin == true ||
        request.auth.uid in ['demonalexander526', 'alexhack235'] ||
        exists(/databases/$(database)/documents/admins/$(request.auth.uid))
      );
    }

    // 4. Group Membership Verification (Safely checks existence, members array, or group creator)
    function isGroupMember(groupId) {
      return isAuthenticated() &&
             exists(/databases/$(database)/documents/groups/$(groupId)) &&
             (
               ('members' in get(/databases/$(database)/documents/groups/$(groupId)).data &&
                request.auth.uid in get(/databases/$(database)/documents/groups/$(groupId)).data.members) ||
               request.auth.uid == get(/databases/$(database)/documents/groups/$(groupId)).data.creatorId ||
               isSuperAdmin()
             );
    }

    // 5. Group Admin Verification (Admins list, group creator, or SuperAdmin)
    function isGroupAdmin(groupId) {
      return isAuthenticated() &&
             exists(/databases/$(database)/documents/groups/$(groupId)) &&
             (
               ('admins' in get(/databases/$(database)/documents/groups/$(groupId)).data &&
                request.auth.uid in get(/databases/$(database)/documents/groups/$(groupId)).data.admins) ||
               request.auth.uid == get(/databases/$(database)/documents/groups/$(groupId)).data.creatorId ||
               isSuperAdmin()
             );
    }

    // 6. Group Moderator Verification (Admin OR Moderator)
    function isGroupModerator(groupId) {
      return isAuthenticated() &&
             exists(/databases/$(database)/documents/groups/$(groupId)) &&
             (
               isGroupAdmin(groupId) ||
               (
                 'moderators' in get(/databases/$(database)/documents/groups/$(groupId)).data &&
                 request.auth.uid in get(/databases/$(database)/documents/groups/$(groupId)).data.moderators
               )
             );
    }

    // 7. String Validation Helper (Non-empty with length limits)
    function validStringLength(field, minLen, maxLen) {
      return field in request.resource.data &&
             request.resource.data[field] is string &&
             request.resource.data[field].size() >= minLen &&
             request.resource.data[field].size() <= maxLen;
    }

    // 8. Optional String Validation Helper (If present, must be string within limits)
    function isValidOptionalString(field, minLen, maxLen) {
      return !(field in request.resource.data) ||
             request.resource.data[field] == null ||
             (request.resource.data[field] is string &&
              request.resource.data[field].size() >= minLen &&
              request.resource.data[field].size() <= maxLen);
    }

    // 9. URL Format Validation
    function isValidUrl(url) {
      return url is string &&
             (
               (url.size() <= 2048 && (url.matches("^https?://.*") || url.matches("^/.*") || url == "")) ||
               (url.size() <= 1048576 && (url.matches("^data:image/.*") || url.matches("^data:video/.*") || url.matches("^blob:.*")))
             );
    }

    // 10. Optional URL Validation
    function isValidOptionalUrl(field) {
      return !(field in request.resource.data) ||
             request.resource.data[field] == null ||
             isValidUrl(request.resource.data[field]);
    }

    // 11. List Size Validation Helper
    function isValidList(val, maxSize) {
      return val is list && val.size() <= maxSize;
    }

    // 12. Optional List Size Validation Helper
    function isValidOptionalList(field, maxSize) {
      return !(field in request.resource.data) ||
             request.resource.data[field] == null ||
             isValidList(request.resource.data[field], maxSize);
    }

    // 12b. Optional Reactions Validation Helper (Supports both array [] and map/object {} representations up to 100 items)
    function isValidOptionalReactions(field) {
      return !(field in request.resource.data) ||
             request.resource.data[field] == null ||
             (request.resource.data[field] is list && request.resource.data[field].size() <= 100) ||
             (request.resource.data[field] is map && request.resource.data[field].size() <= 100);
    }

    // 13. IP Address List Validation Helper (Enforces array/list cap: max 4 IP addresses, each max 45 chars string)
    function isValidIpList(field, maxCount) {
      return !(field in request.resource.data) ||
             request.resource.data[field] == null ||
             (request.resource.data[field] is list &&
              request.resource.data[field].size() <= maxCount &&
              (request.resource.data[field].size() < 1 || (request.resource.data[field][0] is string && request.resource.data[field][0].size() <= 45)) &&
              (request.resource.data[field].size() < 2 || (request.resource.data[field][1] is string && request.resource.data[field][1].size() <= 45)) &&
              (request.resource.data[field].size() < 3 || (request.resource.data[field][2] is string && request.resource.data[field][2].size() <= 45)) &&
              (request.resource.data[field].size() < 4 || (request.resource.data[field][3] is string && request.resource.data[field][3].size() <= 45)));
    }

    // 14. Immutable Field Check
    function areImmutableFieldsUnchanged(fields) {
      return !request.resource.data.diff(resource.data).affectedKeys().hasAny(fields);
    }

    // 15. Extension Hub Scopes Validator (Max 10 scopes, each string max 64 chars)
    function isValidScopeList(scopes) {
      return scopes is list &&
             scopes.size() <= 10 &&
             (scopes.size() < 1 || (scopes[0] is string && scopes[0].size() <= 64)) &&
             (scopes.size() < 2 || (scopes[1] is string && scopes[1].size() <= 64)) &&
             (scopes.size() < 3 || (scopes[2] is string && scopes[2].size() <= 64)) &&
             (scopes.size() < 4 || (scopes[3] is string && scopes[3].size() <= 64)) &&
             (scopes.size() < 5 || (scopes[4] is string && scopes[4].size() <= 64)) &&
             (scopes.size() < 6 || (scopes[5] is string && scopes[5].size() <= 64)) &&
             (scopes.size() < 7 || (scopes[6] is string && scopes[6].size() <= 64)) &&
             (scopes.size() < 8 || (scopes[7] is string && scopes[7].size() <= 64)) &&
             (scopes.size() < 9 || (scopes[8] is string && scopes[8].size() <= 64)) &&
             (scopes.size() < 10 || (scopes[9] is string && scopes[9].size() <= 64));
    }

    // 16. Flexible Timestamp / String / Number validator helper
    function isValidTimestampOrNumber(val) {
      return val == null || val is timestamp || val is string || val is int || val is float || val is number;
    }

    // 17. Resource Existence Helper
    function resourceExists() {
      return resource != null;
    }

    // 18. Chat Room Participant Verification
    function isChatParticipant(chatId) {
      return isAuthenticated() && (
        (
          exists(/databases/$(database)/documents/chats/$(chatId)) &&
          (
            ('participants' in get(/databases/$(database)/documents/chats/$(chatId)).data &&
             request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants) ||
            ('members' in get(/databases/$(database)/documents/chats/$(chatId)).data &&
             request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.members) ||
            ('userIds' in get(/databases/$(database)/documents/chats/$(chatId)).data &&
             request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.userIds) ||
            ('userId' in get(/databases/$(database)/documents/chats/$(chatId)).data &&
             get(/databases/$(database)/documents/chats/$(chatId)).data.userId == request.auth.uid) ||
            ('creatorId' in get(/databases/$(database)/documents/chats/$(chatId)).data &&
             get(/databases/$(database)/documents/chats/$(chatId)).data.creatorId == request.auth.uid)
          )
        ) ||
        chatId == request.auth.uid ||
        chatId.matches('^' + request.auth.uid + '_[^_]+$') ||
        chatId.matches('^[^_]+_' + request.auth.uid + '$') ||
        chatId.matches('^' + request.auth.uid + '-[^-]+$') ||
        chatId.matches('^[^-]+-' + request.auth.uid + '$') ||
        isSuperAdmin()
      );
    }

    // User Profile Validator: Prevents privilege escalation on isAdmin, isSuperAdmin, role, banned, and tokens; enforces max 4 IPs
    function isValidUserCreate(data) {
      return (!('isAdmin' in data) || data.isAdmin == false || isSuperAdmin()) &&
             (!('isSuperAdmin' in data) || data.isSuperAdmin == false || isSuperAdmin()) &&
             (!('role' in data) || data.role in ['user', 'member', 'client', 'guest'] || isSuperAdmin()) &&
             (!('banned' in data) || data.banned == false || isSuperAdmin()) &&
             (!('tokens' in data) || (data.tokens is number && data.tokens >= 0 && data.tokens <= 2000) || isSuperAdmin()) &&
             isValidOptionalString('username', 0, 100) &&
             isValidOptionalString('displayName', 0, 100) &&
             isValidOptionalString('bio', 0, 1000) &&
             isValidOptionalString('email', 0, 255) &&
             isValidOptionalUrl('photoURL') &&
             isValidOptionalUrl('avatar') &&
             isValidOptionalList('blockedUsers', 500) &&
             isValidOptionalList('mutedUsers', 500) &&

      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && (
        request.auth.uid == request.resource.data.userId ||
        isSuperAdmin()
      );
      allow delete: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        isSuperAdmin()
      );
    }

    match /postComments/{commentId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid;
      allow update, delete: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        isSuperAdmin()
      );
    }

    match /stories/{storyId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid;
      allow update, delete: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        isSuperAdmin()
      );
    }

    match /storyViews/{viewId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.resource.data.viewerId == request.auth.uid;
    }

    match /liveStreams/{streamId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.resource.data.hostId == request.auth.uid;
      allow update, delete: if isAuthenticated() && (
        request.auth.uid == resource.data.hostId ||
        isSuperAdmin()
      );
    }

    match /streamViewers/{viewerId} {
      allow read: if isAuthenticated();
      allow create, update: if isAuthenticated() && (
        request.resource.data.userId == request.auth.uid ||
        viewerId == request.auth.uid ||
        isSuperAdmin()
      );
      allow delete: if isAuthenticated() && (
        resource.data.userId == request.auth.uid ||
        viewerId == request.auth.uid ||
        isSuperAdmin()
      );
    }

    match /streamChat/{messageId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() &&
                       request.resource.data.senderId == request.auth.uid &&
                       validStringLength('text', 1, 1000);
      allow delete: if isAuthenticated() && (
        resource.data.senderId == request.auth.uid ||
        isSuperAdmin()
      );
    }

    match /videoComments/{commentId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid &&
                       validStringLength('text', 1, 1000);
      allow update, delete: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||
        isSuperAdmin()
      );
    }

    match /videoLikes/{likeId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && (
        request.auth.uid == request.resource.data.userId ||
        isSuperAdmin()
      );
      allow delete: if isAuthenticated() && (
        request.auth.uid == resource.data.userId ||

    match /playlists/{playlistId} {

    match /playlistVideos/{videoId} {

      allow read: if isAuthenticated() && (

        request.auth.uid == resource.data.recipientId ||
        isSuperAdmin()
      );
      allow create: if isAuthenticated() && request.resource.data.senderId == request.auth.uid;
      allow update, delete: if isAuthenticated() && (
        request.auth.uid == resource.data.senderId ||
        isSuperAdmin()
      );

    // ===============================================================
    // 21. GROUP AUDIO STAGES
    // ===============================================================
    match /groupStages/{stageId} {
      allow read: if isAuthenticated();
      allow create, update: if isAuthenticated();
      allow delete: if isSuperAdmin() || (isAuthenticated() && resource.data.hostUid == request.auth.uid);
    }

    // ===============================================================
    // 22. P2P SIGNALING CHANNELS
    // ===============================================================
    match /p2pSignaling/{channelId} {
      allow read: if isAuthenticated() && (
        resource.data.from == request.auth.uid ||
        resource.data.to == request.auth.uid ||
        resource.data.initiator == request.auth.uid ||
        resource.data.peer == request.auth.uid ||
        isSuperAdmin()
      );
      allow create: if isAuthenticated() && (
        request.resource.data.from == request.auth.uid ||
        request.resource.data.initiator == request.auth.uid ||
        isSuperAdmin()
      );
      allow update, delete: if isAuthenticated() && (
        resource.data.from == request.auth.uid ||
        resource.data.to == request.auth.uid ||
        resource.data.initiator == request.auth.uid ||
        resource.data.peer == request.auth.uid ||
        isSuperAdmin()
      );
    }

    // ===============================================================
    // 23. BROADCAST CHANNELS & SUBSCRIBERS
    // ===============================================================
    match /channels/{channelId} {
      // Authenticated users can read channels; unauthenticated allowed if channel is public
      allow read: if isAuthenticated() || (resource.data.isPublic == true);

      // Channel creation requires authentication, owner match, valid name length, and restricted isVerified flag
      allow create: if isAuthenticated() &&
                       request.resource.data.creatorId == request.auth.uid &&
                       validStringLength('name', 1, 100) &&
                       (!('isVerified' in request.resource.data) || request.resource.data.isVerified == false || isSuperAdmin());

      // Channel update restricted to creator or SuperAdmin, protects creatorId and restricts verification edits; allows subscriber/follower counts and lastPost updates
      allow update: if isAuthenticated() && (
                       (
                         (resource.data.creatorId == request.auth.uid || isSuperAdmin()) &&
                         request.resource.data.creatorId == resource.data.creatorId &&
                         (!('isVerified' in request.resource.data) || request.resource.data.isVerified == resource.data.isVerified || isSuperAdmin())
                       ) ||
                       (
                         request.resource.data.diff(resource.data).affectedKeys().hasOnly(['subscriberCount', 'followerCount', 'lastPostSnippet', 'lastPostTime'])
                       )
                     );

      // Deletion restricted to channel creator or SuperAdmin
      allow delete: if isAuthenticated() && (resource.data.creatorId == request.auth.uid || isSuperAdmin());

      // Subcollection: match /channels/{channelId}/posts/{postId}
      match /posts/{postId} {
        allow read: if isAuthenticated();
        allow create, update: if isAuthenticated() && (
          (exists(/databases/$(database)/documents/channels/$(channelId)) &&
           get(/databases/$(database)/documents/channels/$(channelId)).data.creatorId == request.auth.uid) ||
          request.resource.data.authorId == request.auth.uid ||
          isSuperAdmin()
        ) && (
          isValidOptionalString('text', 0, 10000) ||
          isValidOptionalString('content', 0, 10000)
        );
        allow delete: if isAuthenticated() && (
          (exists(/databases/$(database)/documents/channels/$(channelId)) &&
           get(/databases/$(database)/documents/channels/$(channelId)).data.creatorId == request.auth.uid) ||
          resource.data.authorId == request.auth.uid ||
          isSuperAdmin()
        );
      }

      // Subcollection: match /channels/{channelId}/subscribers/{userId}
      match /subscribers/{userId} {
        allow read: if isAuthenticated();
        allow create, delete: if isAuthenticated() && request.auth.uid == userId;
      }
    }

    // ------------------------------------------------------------------------
    // PASSWORD RESET VERIFICATION CODES (Zero-Trust 6-Digit TTL Vault)
    // ------------------------------------------------------------------------
    match /password_reset_codes/{docId} {
      allow create: if request.resource.data.keys().hasAll(['email', 'code', 'expiresAt', 'used'])
                    && request.resource.data.code is string
                    && request.resource.data.code.size() == 6
                    && request.resource.data.used == false;
      allow read: if true;
      allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(['used', 'usedAt', 'attempts']);
      allow delete: if isSuperAdmin();
    }

    // ===============================================================
    // 24. ZERO-TRUST DEFAULT DENY BARRIER
    // ===============================================================
    match /{document=**} {
      allow read, write: if false;
    }
  }
